### Supported Custom Functions 

* WriteByte("address", "value");
* WriteShort("address", "value");
* WriteInt("address", "value");
* WriteFloat("address", "value");
* WriteString("address", "value");
* ReadByte("address");
* ReadShort("address");
* ReadInt("address");
* ReadFloat("address");
* ReadString("address");
* RPC("address", "arg1", "arg2", "arg3", "arg4", "arg5", "arg6", "arg7", "arg8" );
